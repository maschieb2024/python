SELECT    convert(date,getdate()) as Datum
		, T1.Produkt as Produkt
		, 'WB_FREMD' as Verlag
		, CONVERT(nvarchar(2), '01') as Typ
		, '' as Storno
		, case when isnull(T1.Stück_tgl,0) > 0 then T1.Stück_tgl else 1 end as Anzahl
		, CONVERT(date, DATEADD("day", 1,GETDATE())) as Gueltigab -- datediff getdate +1
		, CONVERT(date, '9999-12-31') Gueltigbis
-- Donnerstag, 6. Dezember 2018
		, case
			when (T1.Hausname = '' or T1.Hausname  is null) then T1.Zusatz1
			else CONVERT(nvarchar(30), LEFT(T1.Hausname,30)) 
			end as name
--		, CONVERT(nvarchar(30), LEFT(T1.Hausname,30)) as name
		, CONVERT(nvarchar(20), LEFT(T1.Vorname,20)) as Vorname
		, CASE WHEN T1.Anrede IN ('HR', 'Herrn', 'Herr', 'Freiherr', 'Eheleute', 'Familie') THEN 'Herr'
			   WHEN T1.Anrede IN ('FR', 'Freifrau', 'Frau') THEN 'Frau' 
			   ELSE 'Firma'
			   END as Anrede
		, CASE WHEN T1.Anrede IN ('HR', 'FR','Herrn', 'Herr', 'Freiherr', 'Freifrau', 'Frau', 'Eheleute', 'Familie') THEN 1 ELSE 2 END as Rechtsform
		, T1.Strasse as Strasse
		, COALESCE(CONCAT(T1.[Haus-Nr], T1.[Haus-Bs]), '') as Hausnummer
		, T1.Plz as PLZ
		, T1.Ort as Ort
		, '38003' as MAId
		, '2' as MAPos
		, CONVERT(nvarchar(4), ' ') as AbGrund
		, CASE WHEN (T1.Stück_tgl IS NULL or T1.Stück_tgl = 0) 
			THEN
			REPLACE(REPLACE(CONCAT(	CASE WHEN ISNULL(T1.Stück_Mo,0) > 0 THEN 1 ELSE 0 END ,
									CASE WHEN ISNULL(T1.Stück_Di,0) > 0 THEN 1 ELSE 0 END ,
									CASE WHEN ISNULL(T1.Stück_Mi,0) > 0 THEN 1 ELSE 0 END ,
									CASE WHEN ISNULL(T1.Stück_Do,0) > 0 THEN 1 ELSE 0 END ,
									CASE WHEN ISNULL(T1.Stück_Fr,0) > 0 THEN 1 ELSE 0 END ,
									CASE WHEN ISNULL(T1.Stück_Sa,0) > 0 THEN 1 ELSE 0 END , 0), '0',' '),'1','X')
			ELSE  'XXXXXX ' END AS eweise
		, T2.[NW-Kd-Nr] as 'GP'
		, format(convert(int,t2.[NW-Verkehrs-Nr]),'000') as 'Adresse'
		, concat(T1.[WB-Kd-Nr],T1.[WB-Abo-Nr]) as VERTRAG
		--, ID_WB
INTO #Zugaenge
from HKABOS as T1
	left join temp_HKABOS_VIVA as T2
	on concat([T1].[WB-Kd-Nr],[T1].[WB-Abo-Nr]) = [T2].[WB-Kd-Nr] 
	where T2.[WB-Kd-Nr] is null

delete from #Zugaenge where (Vertrag is null or Vertrag = '')
